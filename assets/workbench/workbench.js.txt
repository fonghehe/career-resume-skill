function status(text, error = false) { $('status').textContent = text; $('status').classList.toggle('error', error); }
function run(action) { return async () => { try { await action(); } catch (error) { status(error.message, true); } }; }
async function api(path, payload, binary = false) {
  const response = await fetch('/api/' + path, {method: payload === undefined ? 'GET' : 'POST',
    headers: {'X-Workbench-Token': token, 'Content-Type':'application/json'},
    body: payload === undefined ? undefined : JSON.stringify(payload)});
  if (!response.ok) throw new Error((await response.json()).error || '操作失败');
  return binary ? response.blob() : response.json();
}
function option(select, value, text) { const item = new Option(text, value); select.add(item); }
function node(tag, text, className) { const item = document.createElement(tag); item.textContent = text; if (className) item.className = className; return item; }
function keyFor(collection, id) { return collection + '/' + id; }
function record(id = selected, collection = tab) {
  if (collection === 'profile') return saved.data.profile;
  return saved.data[collection].find(row => row.id === id) || {};
}
function rawFields(id, collection) {
  const path = collection === 'projects' ? saved.cards[id] : 'references/profile.md';
  const text = saved.notes[path] || '';
  const match = /<!-- career-workbench-current -->\n```json\n([\s\S]*?)\n```/.exec(text);
  if (!match) return {};
  const group = JSON.parse(match[1])[keyFor(collection, id)] || {};
  return Object.fromEntries(Object.entries(group).map(([field, value]) => [field, value.value]));
}
function draftKey() { return 'career-workbench-draft:' + saved.workspace; }
function remember() {
  try { localStorage.setItem(draftKey(), JSON.stringify({revision:saved.revision,changes:[...changes],notes:[...notes],resume: resumeDirty ? {body:$('resumeBody').value,report:$('resumeReport').value,version:$('resumeVersion').value,language:$('resumeLanguage').value,theme:$('resumeTheme').value} : null})); } catch { /* File saving remains available when browser storage is disabled. */ }
  status(changes.size || notes.size ? `有 ${changes.size + notes.size} 项未保存修改` : '已保存到本地资料库');
  clearTimeout(previewTimer); previewTimer = setTimeout(() => updatePreview().catch(error => status(error.message, true)), 350);
}
function renderRecord() {
  const picker = $('recordPicker'); picker.replaceChildren();
  const collection = tab;
  $('recordTitle').textContent = {profile:'个人资料',career:'工作经历',projects:'项目与案例'}[tab];
  $('addRecord').hidden = tab === 'profile'; $('recordPickerLabel').hidden = tab === 'profile';
  const rows = tab === 'profile' ? [] : saved.data[tab];
  rows.forEach(row => option(picker,row.id,row.id + ' · ' + (row.name || row.organization)));
  for (const change of changes.values()) if (change.collection === tab && !rows.some(row => row.id === change.id)) option(picker,change.id,change.id + ' · 新记录');
  if (tab === 'profile') selected = 'U01';
  else if (![...picker.options].some(item => item.value === selected)) selected = picker.options[0]?.value || '';
  picker.value = selected;
  const form = $('recordForm'); form.replaceChildren();
  if (!selected) { form.append(node('p','还没有记录，点击「新增」开始。')); return; }
  const original = {...record(), ...rawFields(selected, collection)};
  const change = changes.get(keyFor(collection, selected));
  const values = {...original, ...(change?.fields || {})};
  for (const [field, label] of fields[tab]) {
    const wrapper = node('label',label);
    const multiline = ['summary','milestones'].includes(field) || lists.has(field);
    const input = document.createElement(field === 'nature' ? 'select' : multiline ? 'textarea' : 'input');
    input.name = field;
    if (field === 'nature') { option(input,'company','单位或机构实践'); option(input,'personal','独立实践'); }
    let value = values[field] ?? '';
    if (lists.has(field)) value = (value || []).join('\n');
    if (field === 'milestones') value = (value || []).map(item => [item.date || '',item.title,item.summary || ''].join(' | ')).join('\n');
    input.value = value;
    if (multiline) input.rows = field === 'summary' ? 4 : 3;
    if (lists.has(field)) input.placeholder = '每行一项';
    if (field === 'milestones') input.placeholder = '日期 | 阶段标题 | 说明，每行一个阶段';
    if (['start','end'].includes(field)) input.placeholder = 'YYYY、YYYY-MM 或 YYYY-MM-DD；未知留空';
    input.addEventListener('input', collectRecord);
    wrapper.append(input); form.append(wrapper);
    const controls = document.createElement('details'); controls.className = 'field-controls'; controls.append(node('summary','确认状态与事实编号（可选）'));
    const confirmLabel = node('label','', 'check'), confirmInput = document.createElement('input');
    confirmInput.type = 'checkbox'; confirmInput.dataset.confirmField = field;
    confirmInput.checked = Boolean(change?.confirmFields?.includes(field));
    confirmInput.addEventListener('change', collectRecord);
    confirmLabel.append(confirmInput, document.createTextNode('确认此字段，即使文字未改'));
    const idLabel = node('label','已有事实编号（可选）'), factId = document.createElement('input');
    factId.dataset.factField = field; factId.value = change?.factIds?.[field] || '';
    factId.placeholder = collection === 'projects' ? selected + '-A' : 'E01-T';
    factId.addEventListener('input', collectRecord); idLabel.append(factId);
    controls.append(confirmLabel,idLabel); form.append(controls);
  }
  $('editMode').value = change?.mode || 'fact'; $('confirmFact').checked = change?.confirmed || false;
}
function collectRecord() {
  if (!selected) return;
  const collection = tab, id = selected, original = {...record(),...rawFields(id,collection)}, patch = {}, factIds = {}, confirmFields = [];
  const isNew = collection !== 'profile' && !saved.data[collection].some(row => row.id === id);
  for (const [field] of fields[collection]) {
    const input = $('recordForm').elements.namedItem(field); if (!input) continue;
    let value = input.value;
    if (lists.has(field)) value = value.split('\n').map(line => line.trim()).filter(Boolean);
    if (field === 'milestones') value = value.split('\n').filter(line => line.trim()).map(line => {
      const [date, title, ...rest] = line.split('|').map(item => item.trim());
      return {date:date || '',title:title || '',summary:rest.join(' | ')};
    });
    const before = original[field] ?? (lists.has(field) || field === 'milestones' ? [] : field === 'nature' ? 'company' : '');
    const confirmInput = $('recordForm').querySelector(`[data-confirm-field="${field}"]`);
    const factInput = $('recordForm').querySelector(`[data-fact-field="${field}"]`);
    if (confirmInput?.checked) confirmFields.push(field);
    if (factInput?.value.trim()) factIds[field] = factInput.value.trim();
    if (isNew || confirmInput?.checked || JSON.stringify(value) !== JSON.stringify(before)) patch[field] = value;
  }
  const key = keyFor(collection,id);
  if (Object.keys(patch).length) changes.set(key,{collection,id,fields:patch,factIds,confirmFields,confirmOnly:confirmFields.length > 0,mode:$('editMode').value,confirmed:$('confirmFact').checked});
  else changes.delete(key);
  remember();
}
function addRecord() {
  const used = new Set([...saved.data[tab].map(row => row.id), ...[...changes.values()].filter(c => c.collection === tab).map(c => c.id)]);
  let count = 1, id; do { id = (tab === 'career' ? 'E' : 'P') + String(count++).padStart(2,'0'); } while (used.has(id));
  selected = id; changes.set(keyFor(tab,id),{collection:tab,id,fields:{},mode:'fact',confirmed:false}); renderRecord(); remember();
}

function checkboxes(target) {
  const container = $(target); container.replaceChildren();
  const groups = [
    ['careers', '任职与实践', saved.data.career.map(row => [row.id,[row.organization,row.role].filter(Boolean).join(' · ')])],
    ['projects', '代表案例（可选）', saved.data.projects.map(row => [row.id,row.name])],
    ['sections', '教育、资格与其他模块', (saved.data.presentation?.sections || []).filter(row => row.items.length).map(row => [row.id,row.title])],
    ['records', '源记录', Object.keys(saved.notes).filter(path => !path.startsWith('references/projects/')).sort().map(path => [path,path])]
  ];
  for (const [group,title,rows] of groups) {
    if (!rows.length) continue;
    container.append(node('h4',title));
    for (const [id,title] of rows) {
      const label = node('label','', 'check'), input = document.createElement('input');
      input.type = 'checkbox'; input.value = id; input.dataset.group = group;
      label.append(input,document.createTextNode(title));
      if (target === 'projectSelection') {
        const up = node('button','上移'), down = node('button','下移');
        up.type = down.type = 'button';
        up.onclick = () => { if (label.previousElementSibling?.tagName === 'LABEL') container.insertBefore(label,label.previousElementSibling); };
        down.onclick = () => { if (label.nextElementSibling?.tagName === 'LABEL') container.insertBefore(label.nextElementSibling,label); };
        label.append(up,down);
      }
      container.append(label);
    }
  }
}

  tab = value;
  document.querySelectorAll('[data-tab]').forEach(button => button.classList.toggle('active',button.dataset.tab === tab));
  for (const name of ['notes','pending','history','resume','agent']) $(name + 'Panel').hidden = name !== tab;
  $('recordPanel').hidden = !fields[tab];

  if (tab === 'notes') renderNotes();

async function accept(value) {

function normalizedDate(value, end) {
  if (/^\d{4}$/.test(value)) return `${value}-${end ? '12-31' : '01-01'}`;
  if (!/^\d{4}-\d{2}$/.test(value)) return value;
  const [year,month] = value.split('-').map(Number);
  return `${value}-${end ? String(new Date(year,month,0).getDate()).padStart(2,'0') : '01'}`;
}
async function updatePreview() {
  const sequence = ++previewSequence, data = structuredClone(saved.data);
  for (const change of changes.values()) {
    let row = change.collection === 'profile' ? data.profile : data[change.collection].find(item => item.id === change.id);
    if (!row) {
      row = change.collection === 'projects' ? {id:change.id,name:'新项目',organization:'',nature:'company',category:'',status:'confirmed',tech:[],highlights:[],start:'',end:''} : {id:change.id,organization:'待填写单位或任职归属',role:'',summary:'',start:'',end:''};
      data[change.collection].push(row);
    }
    Object.assign(row,change.fields);
    if (change.fields.start !== undefined) row.start = normalizedDate(change.fields.start,false);
    if (change.fields.end !== undefined) row.end = normalizedDate(change.fields.end,true);
  }
  const result = await api('preview',{data,notes:[...notes].map(([path,text]) => ({path,text}))});
  if (sequence !== previewSequence) return;
  $('preview').srcdoc = result.html;
  $('previewLabel').textContent = changes.size || notes.size ? '尚未保存的草稿预览' : '已保存';
}

}
